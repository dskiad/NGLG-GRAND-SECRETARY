def _resolve_appointees(c,registry_no,full_name,honorific,former_office,email,mobile,office_key):
    n=max(len(registry_no),len(full_name),len(honorific),len(former_office),len(email),len(mobile),len(office_key));vals=[]
    def at(arr,i):return (arr[i] if i<len(arr) else '').strip()
    for i in range(n):
        reg=at(registry_no,i);name=at(full_name,i);hon=at(honorific,i) or 'Αδ.';former=at(former_office,i);em=at(email,i);mob=at(mobile,i);okey=at(office_key,i)
        if not any([reg,name,em,mob,okey]):continue
        if reg:
            ex=c.execute('SELECT * FROM members WHERE registry_no=?',(reg,)).fetchone()
            if ex:
                ex=dict(ex);name=name or ex['full_name'];hon=hon or ex['honorific'];former=former or ex['former_office'];em=em or ex['email'];mob=mob or ex['mobile']
        office=DECREE_OFFICE_BY_KEY.get(okey)
        if not office or not office.get('appointable'):raise HTTPException(400,f'Μη έγκυρο ή μη διοριζόμενο αξίωμα στη γραμμή {i+1}.')
        if not name or not em or not mob:raise HTTPException(400,f'Λείπουν στοιχεία στη γραμμή {i+1}. Απαιτούνται ονοματεπώνυμο, email και κινητό.')
        if hon not in HONORIFIC_OPTIONS:hon='Αδ.'
        member_id=_upsert_member(c,reg,name,hon,former,em,mob) if reg else None
        vals.append({'member_id':member_id,'registry_no':reg,'full_name':name,'honorific':hon,'former_office':former,'email':em,'mobile':mob,'office_key':okey,'office_title':office['title'],'appointment_rules':json.dumps(office['rules'],ensure_ascii=False)})
    if not vals:raise HTTPException(400,'Πρέπει να επιλεγεί τουλάχιστον ένας διοριζόμενος Αδελφός.')
    return vals

@app.post('/decree/new')
def decree_create(req:Request,about_text:str=Form(''),registry_no:list[str]=Form(default=[]),full_name:list[str]=Form(default=[]),honorific:list[str]=Form(default=[]),former_office:list[str]=Form(default=[]),email:list[str]=Form(default=[]),mobile:list[str]=Form(default=[]),office_key:list[str]=Form(default=[])):
    u=need(req);today=date.today();ts=now()
    with con() as c:
        pseq,pyear,prot=nextprot(c);dno=next_decree_no(c);vals=_resolve_appointees(c,registry_no,full_name,honorific,former_office,email,mobile,office_key)
        about=(about_text or '').strip()
        if not about:
            titles=list(dict.fromkeys(v['office_title'] for v in vals));about=('διορισμού '+titles[0]) if len(titles)==1 else 'διορισμού Μεγάλων Αξιωματικών'
        plain=decree_body_plain(about,vals,today.isoformat());subject=f"ΔΙΑΤΑΓΜΑ υπ’ αριθμ. {dno}/{today.year} – {about}"
        recipients=', '.join(v['full_name'] for v in vals);recipient_emails=', '.join(dict.fromkeys(v['email'] for v in vals if v['email']))
        cur=c.execute('INSERT INTO letters(protocol_seq,protocol_year,protocol_no,letter_date,subject,body,template_id,recipient_name,recipient_email,status,created_by,signer,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?)',(pseq,pyear,prot,today.isoformat(),subject,plain,None,recipients,recipient_emails,'draft',u['email'],'dimitrios',ts,ts));lid=cur.lastrowid
        c.execute('INSERT INTO decrees(letter_id,decree_no,decree_year,protocol_no,decree_date,about_text,status,created_by,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?,?,?)',(lid,dno,today.year,prot,today.isoformat(),about,'draft',u['email'],ts,ts))
        did=c.execute('SELECT id FROM decrees WHERE decree_no=?',(dno,)).fetchone()['id']
        for i,v in enumerate(vals):
            c.execute('INSERT INTO decree_appointees(decree_id,sort_order,member_id,registry_no,full_name,honorific,former_office,email,mobile,office_key,office_title,appointment_rules,created_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?)',(did,i,v['member_id'],v['registry_no'],v['full_name'],v['honorific'],v['former_office'],v['email'],v['mobile'],v['office_key'],v['office_title'],v['appointment_rules'],ts))
        c.commit()
    return RedirectResponse(f'/decree/{did}',303)
