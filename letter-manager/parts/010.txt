@app.post('/decree/edit/{did}')
def decree_save_edit(req:Request,did:int,about_text:str=Form(''),registry_no:list[str]=Form(default=[]),full_name:list[str]=Form(default=[]),honorific:list[str]=Form(default=[]),former_office:list[str]=Form(default=[]),email:list[str]=Form(default=[]),mobile:list[str]=Form(default=[]),office_key:list[str]=Form(default=[])):
    u=need(req);ts=now()
    with con() as c:
        d=c.execute('SELECT * FROM decrees WHERE id=?',(did,)).fetchone()
        if not d:raise HTTPException(404)
        vals=_resolve_appointees(c,registry_no,full_name,honorific,former_office,email,mobile,office_key);about=(about_text or '').strip()
        if not about:
            titles=list(dict.fromkeys(v['office_title'] for v in vals));about=('διορισμού '+titles[0]) if len(titles)==1 else 'διορισμού Μεγάλων Αξιωματικών'
        c.execute('DELETE FROM decree_appointees WHERE decree_id=?',(did,))
        for i,v in enumerate(vals):
            c.execute('INSERT INTO decree_appointees(decree_id,sort_order,member_id,registry_no,full_name,honorific,former_office,email,mobile,office_key,office_title,appointment_rules,created_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?)',(did,i,v['member_id'],v['registry_no'],v['full_name'],v['honorific'],v['former_office'],v['email'],v['mobile'],v['office_key'],v['office_title'],v['appointment_rules'],ts))
        plain=decree_body_plain(about,vals,d['decree_date']);subject=f"ΔΙΑΤΑΓΜΑ υπ’ αριθμ. {d['decree_no']}/{d['decree_year']} – {about}"
        recipients=', '.join(v['full_name'] for v in vals);recipient_emails=', '.join(dict.fromkeys(v['email'] for v in vals if v['email']))
        c.execute('UPDATE decrees SET about_text=?,updated_at=? WHERE id=?',(about,ts,did));c.execute('UPDATE letters SET subject=?,body=?,recipient_name=?,recipient_email=?,updated_at=? WHERE id=?',(subject,plain,recipients,recipient_emails,ts,d['letter_id']))
    return RedirectResponse(f'/decree/{did}',303)
