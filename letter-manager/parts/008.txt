def decree_rule_matrix_html():
    groups=[];seen=set()
    for o in DECREE_OFFICES:
        if o['group'] not in seen:
            seen.add(o['group']);groups.append(o['group'])
    rows=[]
    for grp in groups:
        rows.append(f"<tr class='rule-group-row'><td colspan='3'>{esc(grp)}</td></tr>")
        for o in [x for x in DECREE_OFFICES if x['group']==grp]:
            status='Διαθέσιμο για Διάταγμα Μεγάλου Διδασκάλου' if o.get('appointable') else esc(o.get('note','Δεν είναι κεντρικός διορισμός'))
            cls='' if o.get('appointable') else ' class="rule-disabled"'
            rows.append(f"<tr{cls}><td>{esc(o['title'])}</td><td><b>{esc(' & '.join('Καν. '+r for r in o['rules']))}</b></td><td>{status}</td></tr>")
    return "<div class='rule-table-wrap'><table><tr><th>Αξίωμα / Περίπτωση</th><th>Κανόνας διορισμού</th><th>Έλεγχος</th></tr>"+''.join(rows)+"</table></div>"

def decree_form_script():
    office_json=json.dumps({k:{'title':v['title'],'rules':v['rules']} for k,v in DECREE_OFFICE_BY_KEY.items() if v.get('appointable')},ensure_ascii=False)
    card_template=decree_appointee_card(999,{}).replace('data-index="999"','data-index="__IDX__"').replace('Αδελφός <span class="row-number">1000','Αδελφός <span class="row-number">__NUM__')
    template_json=json.dumps(card_template,ensure_ascii=False)
    return f"""<script>
const DECREE_OFFICES={office_json};
const APPOINTEE_TEMPLATE={template_json};
function renumberRows(){{
  document.querySelectorAll('.appointee-row').forEach((r,i)=>{{r.dataset.index=i;const n=r.querySelector('.row-number');if(n)n.textContent=i+1;}});
}}
function addAppointee(){{
  const box=document.getElementById('appointees');const i=box.querySelectorAll('.appointee-row').length;
  let h=APPOINTEE_TEMPLATE.replaceAll('__IDX__',String(i)).replaceAll('__NUM__',String(i+1));box.insertAdjacentHTML('beforeend',h);renumberRows();
}}
function removeAppointee(btn){{
  const rows=document.querySelectorAll('.appointee-row');if(rows.length<=1){{alert('Το Διάταγμα πρέπει να έχει τουλάχιστον έναν διοριζόμενο Αδελφό.');return;}}
  btn.closest('.appointee-row').remove();renumberRows();suggestAbout();
}}
async function lookupMember(inp){{
  const reg=(inp.value||'').trim();if(!reg)return;
  const row=inp.closest('.appointee-row');const note=row.querySelector('.member-found');
  try{{
    const res=await fetch('/api/member/'+encodeURIComponent(reg),{{headers:{{'Accept':'application/json'}}}});
    if(!res.ok){{note.style.display='none';return;}}
    const m=await res.json();if(!m.found){{note.style.display='none';return;}}
    row.querySelector('[name=full_name]').value=m.full_name||'';row.querySelector('[name=honorific]').value=m.honorific||'Αδ.';
    row.querySelector('[name=former_office]').value=m.former_office||'';row.querySelector('[name=email]').value=m.email||'';row.querySelector('[name=mobile]').value=m.mobile||'';
    note.style.display='block';
  }}catch(e){{note.style.display='none';}}
}}
function officeChanged(sel){{
  const o=DECREE_OFFICES[sel.value],row=sel.closest('.appointee-row'),d=row.querySelector('.rule-display');d.textContent=o?('Κανόνας: '+o.rules.join(' + ')):'';
  suggestAbout();
}}
function suggestAbout(){{
  const about=document.getElementById('about_text');if(!about || about.dataset.touched==='1')return;
  const titles=[...document.querySelectorAll('.office-select')].map(s=>DECREE_OFFICES[s.value]?.title).filter(Boolean);const unique=[...new Set(titles)];
  if(unique.length===1)about.value='διορισμού '+unique[0];else if(unique.length>1)about.value='διορισμού Μεγάλων Αξιωματικών';
}}
document.addEventListener('DOMContentLoaded',()=>{{
  const about=document.getElementById('about_text');if(about)about.addEventListener('input',()=>about.dataset.touched='1');
  document.querySelectorAll('.office-select').forEach(officeChanged);renumberRows();
}});
</script>"""

    need(req);m=member_by_registry(registry_no)
