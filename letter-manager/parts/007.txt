def next_decree_no(c):
    row=c.execute('SELECT COALESCE(MAX(decree_no),509)+1 n FROM decrees').fetchone()
    return int(row['n'])

def greek_long_date(iso_date):
    d=datetime.fromisoformat(str(iso_date)).date() if 'T' in str(iso_date) else date.fromisoformat(str(iso_date))
    return f"{d.day}ην {GREEK_MONTHS_GEN[d.month]} {d.year}"

def decree_rule_phrase(rules):
    rs=[str(x) for x in rules if str(x).strip()]
    if not rs:return ''
    if len(rs)==1:return f"τον Κανόνα {rs[0]}"
    if len(rs)==2:return f"τους Κανόνας {rs[0]} και {rs[1]}"
    return "τους Κανόνας "+", ".join(rs[:-1])+" και "+rs[-1]

def decree_group_intro(rules):
    rs=[str(x) for x in rules if str(x).strip()]
    if len(rs)==1:return f"Συμφώνως προς τον Κανόνα {rs[0]}, τους κάτωθι Αδ.:"
    if len(rs)==2:return f"Συμφώνως προς τους Κανόνας {rs[0]} και {rs[1]}, τους κάτωθι Αδ.:"
    return "Συμφώνως προς τους Κανόνας "+", ".join(rs[:-1])+" και "+rs[-1]+", τους κάτωθι Αδ.:"

def decree_person_text(a):
    bits=[]
    prefix=(a.get('honorific') or 'Αδ.').strip()
    bits.append(f"{prefix} {(a.get('full_name') or '').strip()}".strip())
    if (a.get('former_office') or '').strip():
        bits.append((a.get('former_office') or '').strip())
    if (a.get('registry_no') or '').strip():
        bits.append("Αριθμ. Μητρώου: "+(a.get('registry_no') or '').strip())
    return ", ".join(bits)

def decree_groups(appointees):
    groups=[];seen={}
    for a in appointees:
        key=tuple(json.loads(a['appointment_rules']) if isinstance(a.get('appointment_rules'),str) else a.get('appointment_rules',[]))
        if key not in seen:
            seen[key]=len(groups);groups.append({'rules':list(key),'items':[]})
        groups[seen[key]]['items'].append(a)
    return groups

def decree_all_rules(appointees):
    out=[]
    for a in appointees:
        rs=json.loads(a['appointment_rules']) if isinstance(a.get('appointment_rules'),str) else a.get('appointment_rules',[])
        for r in rs:
            if r not in out:out.append(r)
    return out

def decree_body_plain(about_text,appointees,decree_date):
    rules=decree_all_rules(appointees)
    lines=['ΗΜΕΙΣ','Ο Μέγας Διδάσκαλος της Εθνικής Μεγάλης Στοάς της Ελλάδος,','Σεβασμιώτατος Αδελφός Ιωάννης Μπενετάτος,',f"λαβόντες υπ’ όψιν {decree_rule_phrase(rules)}",'του Βιβλίου των Γενικών Νόμων και Κανονισμών,',f"περί {about_text.strip().rstrip(',.')},",'ΔΙΟΡΙΖΟΜΕΝ','']
    letters=['Α','Β','Γ','Δ','Ε','ΣΤ','Ζ','Η','Θ','Ι','ΙΑ','ΙΒ','ΙΓ','ΙΔ','ΙΕ']
    for gi,g in enumerate(decree_groups(appointees)):
        lines.append(f"{letters[gi] if gi<len(letters) else str(gi+1)}. {decree_group_intro(g['rules'])}")
        for a in g['items']:lines.append(decree_person_text(a))
        lines.append('')
    lines.append(f"Εγένετο εν τω Τεκτονικώ Μεγάρω της πόλεως των Αθηνών, την {greek_long_date(decree_date)}.")
    return '
'.join(lines)

def get_decree(c,did):
    d=c.execute('SELECT * FROM decrees WHERE id=?',(did,)).fetchone()
    if not d:return None,[]
    xs=[dict(r) for r in c.execute('SELECT * FROM decree_appointees WHERE decree_id=? ORDER BY sort_order,id',(did,))]
    return dict(d),xs

def decree_office_options(selected=''):
    out=['<option value="">— Επιλογή αξιώματος —</option>']
    for o in DECREE_OFFICES:
        if not o.get('appointable'):continue
        label=f"{o['title']} — {' & '.join('Καν. '+r for r in o['rules'])}"
        out.append(f"<option value='{esc(o['key'])}' {'selected' if selected==o['key'] else ''}>{esc(label)}</option>")
    return ''.join(out)

def honorific_options(selected='Αδ.'):
    return ''.join(f"<option value='{esc(x)}' {'selected' if x==selected else ''}>{esc(x)}</option>" for x in HONORIFIC_OPTIONS)

def member_by_registry(registry_no):
    reg=(registry_no or '').strip()
    if not reg:return None
    with con() as c:
        r=c.execute('SELECT * FROM members WHERE registry_no=?',(reg,)).fetchone()
        return dict(r) if r else None

def _upsert_member(c,registry_no,full_name,honorific,former_office,email,mobile):
    reg=(registry_no or '').strip()
    if not reg:return None
    existing=c.execute('SELECT * FROM members WHERE registry_no=?',(reg,)).fetchone();ts=now()
    if existing:
        c.execute('UPDATE members SET full_name=?,honorific=?,former_office=?,email=?,mobile=?,updated_at=? WHERE registry_no=?',(full_name,honorific,former_office,email,mobile,ts,reg))
    else:
        c.execute('INSERT INTO members(registry_no,full_name,honorific,former_office,email,mobile,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?)',(reg,full_name,honorific,former_office,email,mobile,ts,ts))
    return c.execute('SELECT id FROM members WHERE registry_no=?',(reg,)).fetchone()['id']
